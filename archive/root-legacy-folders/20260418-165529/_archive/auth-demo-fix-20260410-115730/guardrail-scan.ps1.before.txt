$ErrorActionPreference = "Stop"

$repoRoot = Split-Path -Parent $PSScriptRoot
Set-Location $repoRoot

function Get-LiveFiles {
  param(
    [Parameter(Mandatory = $true)][string[]]$Roots
  )

  Get-ChildItem $Roots -Recurse -File |
    Where-Object {
      $_.FullName -notmatch '\\node_modules\\' -and
      $_.FullName -notmatch '\\dist\\' -and
      $_.FullName -notmatch '\\_QUARANTINE\\' -and
      $_.FullName -notmatch '\\_archive\\' -and
      $_.FullName -notmatch '\\_AUDITS\\' -and
      $_.Name -notmatch '\.bak-' -and
      $_.Name -notmatch '\.backup-' -and
      $_.Name -notmatch '\.before-pass' -and
      $_.Name -notmatch '\.wingman-agents-backup-' -and
      $_.Name -notmatch '\.public-landing-backup-'
    }
}

$violations = New-Object System.Collections.Generic.List[object]

function Add-Violation {
  param(
    [Parameter(Mandatory = $true)][string]$Rule,
    [Parameter(Mandatory = $true)]$Match
  )

  $shortPath = $Match.Path.Substring($repoRoot.Length).TrimStart('\')
  $violations.Add([pscustomobject]@{
    Rule = $Rule
    Path = $shortPath
    Line = $Match.LineNumber
    Text = $Match.Line.Trim()
  })
}

function Find-Forbidden {
  param(
    [Parameter(Mandatory = $true)][string]$Rule,
    [Parameter(Mandatory = $true)][string[]]$Roots,
    [Parameter(Mandatory = $true)][string[]]$Patterns,
    [string[]]$ExcludeRelativePaths = @()
  )

  $excludeSet = @{}
  foreach ($exclude in $ExcludeRelativePaths) {
    $excludeSet[$exclude.ToLowerInvariant()] = $true
  }

  $files = Get-LiveFiles -Roots $Roots

  foreach ($file in $files) {
    $relative = $file.FullName.Substring($repoRoot.Length).TrimStart('\').Replace('/', '\')
    $relativeKey = $relative.ToLowerInvariant()

    if ($excludeSet.ContainsKey($relativeKey)) {
      continue
    }

    $matches = Select-String -Path $file.FullName -SimpleMatch -Pattern $Patterns
    foreach ($match in $matches) {
      Add-Violation -Rule $Rule -Match $match
    }
  }
}

Find-Forbidden `
  -Rule "No hard navigation in live UI" `
  -Roots @(".\src\features", ".\src\pages") `
  -Patterns @("window.location.href =")

Find-Forbidden `
  -Rule "No legacy saved-projects panel in live app" `
  -Roots @(".\src") `
  -Patterns @("WingmanSavedProjectsPanel")

Find-Forbidden `
  -Rule "No legacy generation context in live app" `
  -Roots @(".\src") `

Find-Forbidden `
  -Rule "No legacy persisted project model in live app" `
  -Roots @(".\src") `
  -Patterns @(
    'wingmanProjectPersistence',
    'loadAllPersistedProjects(',
    'setActivePersistedProjectId(',
    'upsertPersistedProject(',
    'buildProposalPersistedProject(',
    'buildVideoWallPersistedProject('
  )

Find-Forbidden `
  -Rule "No proposal local-persist bridge in live app" `
  -Roots @(".\src") `
  -Patterns @('proposal/bom/persist', 'loadProposal(', 'saveProposal(')

Find-Forbidden `
  -Rule "No demo sign-in usage outside auth definition" `
  -Roots @(".\src") `
  -Patterns @('signInDemo(') `
  -ExcludeRelativePaths @(
    'src\context\AuthContext.tsx',
    'src\state\wingmanStore.ts'
  )

if ($violations.Count -eq 0) {
  Write-Host "Guardrail scan passed" -ForegroundColor Green
  exit 0
}

Write-Host ""
Write-Host "Guardrail scan failed" -ForegroundColor Red
Write-Host ""

$grouped = $violations | Group-Object Rule
foreach ($group in $grouped) {
  Write-Host $group.Name -ForegroundColor Yellow
  foreach ($item in $group.Group | Sort-Object Path, Line) {
    Write-Host ("  {0}:{1}: {2}" -f $item.Path, $item.Line, $item.Text)
  }
  Write-Host ""
}

exit 1
