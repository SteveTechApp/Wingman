import { useCallback, useEffect, useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import { routeMap } from "@/core/wingman/routeMap";
import { useAuth } from "@/context/AuthContext";
import "./wingman-public-landing.css";

const JOURNEY = [
  {
    step: "01",
    title: "Discovery",
    copy: "Capture the room, scope, and key constraints.",
  },
  {
    step: "02",
    title: "Design",
    copy: "Narrow the right WyreStorm solution path quickly.",
  },
  {
    step: "03",
    title: "Proposal",
    copy: "Move into a clean handoff once the direction is stable.",
  },
];

export default function PublicLandingPage() {
  const navigate = useNavigate();
  const { signInDemo } = useAuth();
  const [demoSubmitting, setDemoSubmitting] = useState(false);

  useEffect(() => {
    document.documentElement.setAttribute("data-wm-public-landing-page", "true");
    document.body.setAttribute("data-wm-public-landing-page", "true");

    return () => {
      document.documentElement.removeAttribute("data-wm-public-landing-page");
      document.body.removeAttribute("data-wm-public-landing-page");
    };
  }, []);

  const handleDemo = useCallback(async () => {
    setDemoSubmitting(true);
    try {
      await signInDemo();
      navigate(routeMap.app.dashboard, { replace: true });
    } finally {
      setDemoSubmitting(false);
    }
  }, [navigate, signInDemo]);

  return (
    <div className="wm-public-landing">
      <main className="wm-public-landing__main">
        <div className="wm-public-landing__brand" aria-label="WyreStorm Wingman">
          <div className="wm-public-landing__brand-mark">
            <span className="wm-public-landing__brand-wyrestorm">WyreStorm</span>
            <span className="wm-public-landing__brand-wingman">Wingman</span>
          </div>
        </div>

        <section className="wm-public-landing__hero">
          <h1 className="wm-public-landing__title">
            One operating surface for product fit, room design, and proposal handoff.
          </h1>

          <p className="wm-public-landing__summary">
            Wingman keeps technical sales work clear, compact, and project focused.
          </p>

          <div className="wm-public-landing__actions">
            <Link className="wm-public-landing__button wm-public-landing__button--ghost" to="/login">
              Log in
            </Link>
            <Link className="wm-public-landing__button wm-public-landing__button--primary" to="/signup">
              Start Wingman
            </Link>
            <button
              type="button"
              className="wm-public-landing__button wm-public-landing__button--ghost"
              onClick={handleDemo}
              disabled={demoSubmitting}
            >
              {demoSubmitting ? "Opening demo..." : "Demo mode"}
            </button>
          </div>
        </section>

        <section className="wm-public-landing__journey" aria-label="Wingman journey">
          {JOURNEY.map((item) => (
            <article key={item.step} className="wm-public-landing__card">
              <span className="wm-public-landing__step">{item.step}</span>
              <h2 className="wm-public-landing__card-title">{item.title}</h2>
              <p className="wm-public-landing__card-copy">{item.copy}</p>
            </article>
          ))}
        </section>
      </main>
    </div>
  );
}
